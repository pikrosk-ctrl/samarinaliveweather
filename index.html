function buildDailyArchive() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName("Φύλλο1") || ss.getSheets()[0]; let archiveSheet = ss.getSheetByName("Daily_Archive"); if (!archiveSheet) { archiveSheet = ss.insertSheet("Daily_Archive"); } const lastRow = sheet.getLastRow(); if (lastRow < 2) return; const tz = "Europe/Athens"; const lastCol = Math.max(sheet.getLastColumn(), 39); const data = sheet.getRange(2, 1, lastRow - 1, lastCol).getValues(); const daysMap = {}; let prevDt = null; for (let i = 0; i < data.length; i++) { const dt = parseGreekDate(data[i][0]); if (!dt) continue; const dayKey = Utilities.formatDate(dt, tz, "yyyy-MM-dd"); const temp = Number(data[i][1]); const precipRate = Number(data[i][2] || 0); const dir = data[i][8] !== "" ? Number(data[i][8]) : null; const press = data[i][9] !== "" ? Number(data[i][9]) : null; const errSt = data[i][10] !== "" ? Number(data[i][10]) : null; const errHyb = data[i][19] !== "" ? Number(data[i][19]) : null; const windSpd = (data[i][21] !== undefined && data[i][21] !== "") ? Number(data[i][21]) : null; const windGst = (data[i][22] !== undefined && data[i][22] !== "") ? Number(data[i][22]) : null; const errHyb3 = (data[i][29] !== undefined && data[i][29] !== "") ? Number(data[i][29]) : null; const errHyb6 = (data[i][37] !== undefined && data[i][37] !== "") ? Number(data[i][37]) : null; let dtHours = 0.5; if (prevDt) { const diff = (dt - prevDt) / (3600 * 1000); if (diff > 0 && diff <= 1.5) dtHours = diff; } prevDt = dt; const precipMm = precipRate * dtHours; if (!daysMap[dayKey]) { daysMap[dayKey] = { date: dayKey, temps: [], precips: [], dirs: [], pressures: [], winds: [], gusts: [], errSt: [], errHyb: [], errHyb3: [], errHyb6: [] }; } if (!isNaN(temp)) daysMap[dayKey].temps.push(temp); daysMap[dayKey].precips.push(precipMm); if (dir !== null && !isNaN(dir)) daysMap[dayKey].dirs.push(dir); if (press !== null && !isNaN(press) && press > 800) daysMap[dayKey].pressures.push(press); if (windSpd !== null && !isNaN(windSpd)) daysMap[dayKey].winds.push(windSpd); if (windGst !== null && !isNaN(windGst)) daysMap[dayKey].gusts.push(windGst); if (errSt !== null && !isNaN(errSt)) daysMap[dayKey].errSt.push(errSt); if (errHyb !== null && !isNaN(errHyb)) daysMap[dayKey].errHyb.push(errHyb); if (errHyb3 !== null && !isNaN(errHyb3)) daysMap[dayKey].errHyb3.push(errHyb3); if (errHyb6 !== null && !isNaN(errHyb6)) daysMap[dayKey].errHyb6.push(errHyb6); } const sortedDays = Object.keys(daysMap).sort(); const rowsOut = [[ "Date", "Samples", "Tmin", "Tmax", "Tavg", "PrecipSum", "DomDirDeg", "DomDirText", "PressMin", "PressMax", "PressAvg", "WindAvg", "WindMax", "MAE_Station", "MAE_Hybrid", "MAE_Hyb_3h", "MAE_Hyb_6h" ]]; const degToCompass = (deg) => { const dirs = ["Β", "ΒΒΑ", "ΒΑ", "ΑΒΑ", "Α", "ΑΝΑ", "ΝΑ", "ΝΝΑ", "Ν", "ΝΝΔ", "ΝΔ", "ΔΝΔ", "Δ", "ΔΒΔ", "ΒΔ", "ΒΒΔ"]; return dirs[Math.round(((deg % 360) / 22.5)) % 16]; }; for (const dayKey of sortedDays) { const d = daysMap[dayKey]; if (!d.temps.length) continue; const tMin = Math.min(...d.temps); const tMax = Math.max(...d.temps); const tAvg = d.temps.reduce((a, b) => a + b, 0) / d.temps.length; const precipSum = d.precips.reduce((a, b) => a + b, 0); let domDeg = ""; let domText = "—"; if (d.dirs.length) { let sinSum = 0, cosSum = 0; d.dirs.forEach(deg => { sinSum += Math.sin(deg * Math.PI / 180); cosSum += Math.cos(deg * Math.PI / 180); }); let meanDeg = Math.round(Math.atan2(sinSum, cosSum) * 180 / Math.PI); if (meanDeg < 0) meanDeg += 360; domDeg = meanDeg; domText = degToCompass(meanDeg); } const pMin = d.pressures.length ? Number(Math.min(...d.pressures).toFixed(1)) : ""; const pMax = d.pressures.length ? Number(Math.max(...d.pressures).toFixed(1)) : ""; const pAvg = d.pressures.length ? Number((d.pressures.reduce((a, b) => a + b, 0) / d.pressures.length).toFixed(1)) : ""; const wAvg = d.winds.length ? Number((d.winds.reduce((a, b) => a + b, 0) / d.winds.length).toFixed(1)) : ""; const allWindPeaks = [...d.winds, ...d.gusts]; const wMax = allWindPeaks.length ? Number(Math.max(...allWindPeaks).toFixed(1)) : ""; const maeSt = d.errSt.length ? Number((d.errSt.reduce((a, b) => a + b, 0) / d.errSt.length).toFixed(2)) : ""; const maeHyb = d.errHyb.length ? Number((d.errHyb.reduce((a, b) => a + b, 0) / d.errHyb.length).toFixed(2)) : ""; const maeHyb3 = d.errHyb3.length ? Number((d.errHyb3.reduce((a, b) => a + b, 0) / d.errHyb3.length).toFixed(2)) : ""; const maeHyb6 = d.errHyb6.length ? Number((d.errHyb6.reduce((a, b) => a + b, 0) / d.errHyb6.length).toFixed(2)) : ""; rowsOut.push([ dayKey, d.temps.length, Number(tMin.toFixed(1)), Number(tMax.toFixed(1)), Number(tAvg.toFixed(1)), Number(precipSum.toFixed(1)), domDeg, domText, pMin, pMax, pAvg, wAvg, wMax, maeSt, maeHyb, maeHyb3, maeHyb6 ]); } archiveSheet.clearContents(); archiveSheet.getRange(1, 1, rowsOut.length, rowsOut[0].length).setValues(rowsOut); }
